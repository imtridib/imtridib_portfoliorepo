const EASE: [number, number, number, number] = [0.22, 1, 0.36, 1];

// Timing (seconds): tweak these to speed up or slow down the two animations
const HEADING_START = 0.2; // delay before the first word
const WORD_STEP = 0.12; // delay between words
const WORD_DURATION = 1.2; // how long each word takes to slide in

const TYPED_SUMMARY =
  "Intelligent solutionsâ€”from Machine learning models to automated workflows. An AI ML & Automation Engineer passionate about building accessible and user friendly websites.";

const TYPE_START = 0.5;
const TYPE_BASE = 45; // ms per character
const TYPE_JITTER = 25; // small variation between characters
const TYPE_CURSOR_LINGER = 1800; // ms the blinking cursor stays after typing finishes (0 = remove at once)

/** "Hi, I am Tridib Chakraborty." revealed one word at a time: slide up + fade in. */
function HeadingWords({ text, reduce }: { text: string; reduce: boolean }) {
  const words = text.split(" ");
  return (
    <>
      {words.map((word, i) => (
        <span key={i} aria-hidden="true">
          <motion.span
            className="inline-block"
            initial={reduce ? false : { opacity: 0, y: "0.6em" }}
            animate={{ opacity: 1, y: 0 }}
            transition={{
              duration: WORD_DURATION,
              delay: HEADING_START + i * WORD_STEP,
              ease: EASE,
            }}
          >
            {word}
          </motion.span>
          {i < words.length - 1 ? " " : ""}
        </span>
      ))}
    </>
  );
}

function TypewriterText({ text, reduce }: { text: string; reduce: boolean }) {
  const [count, setCount] = useState(reduce ? text.length : 0);
  const [cursor, setCursor] = useState(false);

  // Words and phrases to highlight
  const highlights = [
    "Intelligent solutions",
    "Machine learning models",
    "AI ML & Automation Engineer",
  ];

  // Render typed characters with selective highlights
  const renderHighlightedText = () => {
    const parts = [];
    let index = 0;

    while (index < count) {
      const match = highlights.find((word) => text.startsWith(word, index));

      if (match) {
        const visibleText = text.slice(
          index,
          Math.min(index + match.length, count),
        );

        parts.push(
          <strong key={index} className="font-semibold text-accent">
            {visibleText}
          </strong>,
        );

        index += visibleText.length;
      } else {
        const nextHighlight = highlights
          .map((word) => text.indexOf(word, index))
          .filter((position) => position !== -1);

        const nextIndex =
          nextHighlight.length > 0 ? Math.min(...nextHighlight) : count;

        const end = Math.min(nextIndex, count);

        parts.push(text.slice(index, end));
        index = end;
      }
    }

    return parts;
  };

  useEffect(() => {
    if (reduce) {
      setCount(text.length);
      setCursor(false);
      return;
    }

    setCount(0);
    setCursor(false);

    let i = 0;
    let timer = 0;

    const tick = () => {
      i += 1;
      setCount(i);

      if (i >= text.length) {
        timer = window.setTimeout(() => setCursor(false), TYPE_CURSOR_LINGER);
        return;
      }

      const ch = text[i - 1];

      let delay = TYPE_BASE + Math.random() * TYPE_JITTER;

      if (ch === " ") delay += 25;
      if (ch === "," || ch === ".") delay += 220;

      timer = window.setTimeout(tick, delay);
    };

    timer = window.setTimeout(() => {
      setCursor(true);
      tick();
    }, TYPE_START * 1000);

    return () => window.clearTimeout(timer);
  }, [text, reduce]);

  return (
    <>
      <span className="sr-only">{text}</span>

      <span aria-hidden="true">
        {renderHighlightedText()}

        {cursor && (
          <motion.span
            className="relative inline-block h-[1em] w-0 align-[-0.12em]"
            animate={{ opacity: [1, 1, 0, 0] }}
            transition={{
              duration: 1,
              repeat: Infinity,
              times: [0, 0.5, 0.5, 1],
            }}
          >
            <span className="absolute left-[1px] top-0 h-full w-[2px] rounded-full bg-[var(--accent)]" />
          </motion.span>
        )}

        <span className="invisible">{text.slice(count)}</span>
      </span>
    </>
  );
}

import Container from "../ui/Container";
import TiltCard from "../ui/TiltCard";

/* ---------- one-time reveal helpers (local to this file) ---------- */

const EASE = [0.22, 1, 0.36, 1] as const;
// once: true -> triggers the first time only; margin -> fires when ~10% inside the viewport
const VIEW = { once: true, margin: "0px 0px -10% 0px" } as const;

function RevealText({
  children,
  className,
  order = 0,
}: {
  children: ReactNode;
  className?: string;
  order?: number;
}) {
  const reduce = useReducedMotion();
  const ref = useRef<HTMLDivElement>(null);
  const inView = useInView(ref, VIEW);

  if (reduce) return <div className={className}>{children}</div>;

  return (
    <motion.div
      ref={ref}
      className={className}
      initial={{ opacity: 0, y: 36 }}
      animate={inView ? { opacity: 1, y: 0 } : undefined}
      transition={{ duration: 0.8, ease: EASE, delay: order * 0.12 }}
    >
      {children}
    </motion.div>
  );
}

function RevealCard({
  children,
  className,
}: {
  children: ReactNode;
  className?: string;
}) {
  const reduce = useReducedMotion();
  const ref = useRef<HTMLDivElement>(null);
  const inView = useInView(ref, VIEW);

  if (reduce) return <div className={className}>{children}</div>;

  return (
    <motion.div
      ref={ref}
      className={className}
      initial={{ opacity: 0, y: 70, scale: 0.94 }}
      animate={inView ? { opacity: 1, y: 0, scale: 1 } : undefined}
      transition={{ duration: 0.9, ease: EASE }}
    >
      {children}
    </motion.div>
  );
}

function RevealHeadline({ text }: { text: string }) {
  const reduce = useReducedMotion();
  const ref = useRef<HTMLSpanElement>(null);
  const inView = useInView(ref, VIEW);

  if (reduce) return <span>{text}</span>;

  const words = text.split(" ");
  return (
    <span ref={ref} aria-label={text}>
      {words.map((word, i) => (
        <span key={i} aria-hidden="true">
          <span className="inline-block overflow-hidden pb-[0.15em] -mb-[0.15em] align-bottom">
            <motion.span
              className="inline-block"
              initial={{ y: "110%", opacity: 0 }}
              animate={inView ? { y: "0%", opacity: 1 } : undefined}
              transition={{ duration: 0.7, ease: EASE, delay: i * 0.05 }}
            >
              {word}
            </motion.span>
          </span>
          {i < words.length - 1 ? " " : ""}
        </span>
      ))}
    </span>
  );
}

/* ---------- section ---------- */

export default function About() {
  const { summary, paragraphs, degree, institution } = about;
  const [logoFailed, setLogoFailed] = useState(false);

  return (
    <section id="about" className="border-b border-line py-20">
      <Container className="grid items-center gap-12 lg:grid-cols-[1.5fr_1fr]">
        <div>
          {/* plain heading: no animation */}
          <h2 className="font-display text-5xl font-normal sm:text-6xl">
            About me
          </h2>

          {/* summary line: words rise one after another, once */}
          <h3 className="mt-7 font-body text-2xl font-medium leading-snug">
            <RevealHeadline text={summary} />
          </h3>

          {/* paragraphs slide in one after another, once */}
          {/* {paragraphs.map((text, i) => (
            <RevealText key={text} order={i + 1} className="mt-5">
              <p className="text-muted">{text}</p>
            </RevealText>
          ))} */}
          {/* paragraphs slide in one after another, once */}
          {paragraphs.map((text, i) => (
            <RevealText key={i} order={i + 1} className="mt-5">
              <p
                className="text-muted"
                dangerouslySetInnerHTML={{ __html: text }}
              />
            </RevealText>
          ))}

          {degree && (
            <RevealText order={paragraphs.length + 1} className="mt-7">
              <p className="text-accent">{degree}</p>
            </RevealText>
          )}
        </div>

        {/* university card: rises into place once; keeps the tilt + cursor glow on hover */}
        <RevealCard>
          <TiltCard
            max={5}
            className="rounded-2xl border border-line bg-card p-8"
          >
            <a
              href={institution.url}
              target="_blank"
              rel="noreferrer"
              className="relative z-10 block"
            >
              <div className="rounded-xl bg-white p-5">
                {institution.logo && !logoFailed ? (
                  <img
                    src={institution.logo}
                    alt={`${institution.name} official logo`}
                    className="mx-auto h-auto max-h-44 w-full object-contain"
                    loading="lazy"
                    onError={() => setLogoFailed(true)}
                  />
                ) : (
                  // shown until the logo file exists
                  <p className="py-10 text-center font-body text-xl font-semibold text-black">
                    {institution.name}
                  </p>
                )}
              </div>
              <p className="mt-6 text-center text-sm text-muted">
                {institution.name}
                {institution.department && (
                  <>
                    <br />
                    {institution.department}
                  </>
                )}
              </p>
            </a>
          </TiltCard>
        </RevealCard>
      </Container>
    </section>
  );
}
